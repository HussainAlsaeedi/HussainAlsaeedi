WITH customer_value AS (
  SELECT customer_id,
         COUNT(DISTINCT order_id) AS orders,
         SUM(net_revenue) AS revenue
  FROM orders
  GROUP BY customer_id
)
SELECT customer_segment,
       SUM(revenue) AS revenue
FROM customer_value
GROUP BY customer_segment;
